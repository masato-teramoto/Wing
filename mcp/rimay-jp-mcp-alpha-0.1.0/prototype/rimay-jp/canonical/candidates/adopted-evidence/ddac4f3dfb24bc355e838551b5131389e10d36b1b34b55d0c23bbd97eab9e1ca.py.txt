RIMAY_JP = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(RIMAY_JP / "core"))
sys.path.insert(0, str(RIMAY_JP / "scripts"))

from build_response_expression_candidate import (  # noqa: E402
    historical_base_schema, rendered_candidate,
)
from rimay_jp_core.response_expression_candidate import (  # noqa: E402
    RESPONSE_EXPRESSION_CANDIDATE_VERSION,
    ResponseExpressionError,
    build_response_expression_candidate_schema,
    canonical_response_expression_meaning,
    canonical_response_expression_shape,

BASE_SCHEMA = RIMAY_JP / "canonical/canonical-requirement.schema.json"
CANDIDATE_SCHEMA = (
    RIMAY_JP / "canonical/candidates/"
    "canonical-requirement-response-expression-0.6.0.schema.json"
)


def _leaf(index, start, end):
    return {
        "kind": "atomic",
        "span": {"start": start, "end": end},
        "source": f"R{index}",
        "atomic_response_index": index,
    }


def _repository_expression():
    return {
        "kind": "and",
        "span": {"start": 0, "end": 19},
        "source": "(R0 or R1) and R2",
        "operator": {
            "kind": "and", "span": {"start": 11, "end": 14}, "source": "and"},
        "left": {
            "kind": "group",
            "span": {"start": 0, "end": 10},
            "source": "(R0 or R1)",
            "expression": {
                "kind": "or",
                "span": {"start": 1, "end": 9},
                "source": "R0 or R1",
                "operator": {
                    "kind": "or", "span": {"start": 4, "end": 6}, "source": "or"},
                "left": _leaf(0, 1, 3),
                "right": _leaf(1, 7, 9),
            },
        },
        "right": _leaf(2, 15, 17),
    }


def _response(index):
    return {
        "actor": "System",
        "action": {"family": f"action-{index}", "rimay_action_id": None},
        "theme": f"object-{index}",
    }


def _document(requirement_id="REQ-CANDIDATE-1"):
    return {
        "schema_version": "0.5.0",
        "profile_version": "rimay-jp-mvp-0.1.0",
        "requirement_id": requirement_id,
        "revision": 1,
        "status": "candidate",
        "rimay_core": {"response": _response(0)},
        "extensions": {},
        "resolution": {"slots": {}},
        "provenance": [],
    }


def _schemas():
    active = json.loads(BASE_SCHEMA.read_text(encoding="utf-8"))
    base = historical_base_schema(active)
    candidate = json.loads(CANDIDATE_SCHEMA.read_text(encoding="utf-8"))
    return active, base, candidate


def test_candidate_schema_is_deterministic_and_matches_the_promoted_active_shape():
    active, base, candidate = _schemas()
    assert candidate == build_response_expression_candidate_schema(base)
    assert CANDIDATE_SCHEMA.read_text(encoding="utf-8") == rendered_candidate()
    Draft202012Validator.check_schema(candidate)
    assert base["properties"]["schema_version"]["const"] == "0.5.0"
    assert active["properties"]["schema_version"]["const"] == "0.9.0"
    assert candidate["properties"]["schema_version"]["const"] == \
        RESPONSE_EXPRESSION_CANDIDATE_VERSION
    assert "response_expression" not in base["$defs"]["rimayCore"]["properties"]
    assert "response_expression" in candidate["$defs"]["rimayCore"]["properties"]
    for name in (
        "repositoryResponseTrace", "responseExpression", "responseExpressionLeaf",
        "responseExpressionGroup", "responseExpressionAnd", "responseExpressionOr",
    ):
        assert active["$defs"][name] == candidate["$defs"][name]


def test_repository_tree_projects_without_losing_operator_group_or_leaf_order():
    repository = _repository_expression()
    projected = project_repository_response_expression(repository, _response)
    assert response_expression_findings(projected) == ()
    assert repository_response_expression_shape(repository) == \
        canonical_response_expression_shape(projected)
    assert projected["kind"] == "and"
    assert projected["left"]["kind"] == "group"
    assert projected["left"]["expression"]["kind"] == "or"
    assert projected["repository_trace"]["repository_rule_id"] == "AND_Response"
    assert projected["left"]["repository_trace"]["repository_rule_id"] == \
        "PrimaryExpression_Response"
    assert [
        projected["left"]["expression"]["left"]["repository_trace"]
        ["atomic_response_index"],
        projected["left"]["expression"]["right"]["repository_trace"]
        ["atomic_response_index"],
        projected["right"]["repository_trace"]["atomic_response_index"],
    ] == [0, 1, 2]


def test_candidate_schema_accepts_expression_and_rejects_dual_response_sources():
    _, _, candidate = _schemas()
    validator = Draft202012Validator(candidate)
    projected = project_repository_response_expression(_repository_expression(), _response)
    document = _document()
    document["schema_version"] = RESPONSE_EXPRESSION_CANDIDATE_VERSION
    document["rimay_core"] = {"response_expression": projected}
    assert list(validator.iter_errors(document)) == []

    document["rimay_core"]["response"] = _response(9)
    errors = list(validator.iter_errors(document))
    assert errors
    assert response_core_findings(document["rimay_core"])[0]["code"] == \
        "RESPONSE_EXPRESSION_DUAL_SOURCE"


def test_structured_diagnostics_reject_duplicate_leaf_and_trace_tampering():
    projected = project_repository_response_expression(_repository_expression(), _response)
    changed = copy.deepcopy(projected)
    changed["right"]["repository_trace"]["atomic_response_index"] = 0
    changed["left"]["repository_trace"]["source_span"] = {"start": 30, "end": 40}
    codes = {item["code"] for item in response_expression_findings(changed)}
    assert "RESPONSE_EXPRESSION_DUPLICATE_LEAF" in codes
    assert "RESPONSE_EXPRESSION_TRACE_OUTSIDE_PARENT" in codes


def test_projection_rejects_duplicate_repository_leaf_before_callback_reuse():
    repository = _repository_expression()
    repository["right"]["atomic_response_index"] = 0
    with pytest.raises(ResponseExpressionError) as caught:
        project_repository_response_expression(repository, _response)
    assert caught.value.code == "RESPONSE_EXPRESSION_DUPLICATE_LEAF"


def test_projection_rejects_operator_kind_mismatch():
    repository = _repository_expression()

        project_repository_response_expression(repository, _response)
    assert caught.value.code == "RESPONSE_EXPRESSION_OPERATOR_MISMATCH"


def test_english_and_japanese_rendering_make_the_same_tree_explicit():
    projected = project_repository_response_expression(_repository_expression(), _response)

    def render_leaf(response):
        return response["action"]["family"]

    english = render_response_expression(projected, render_leaf, language="en")
    japanese = render_response_expression(projected, render_leaf, language="ja")
    assert english == "(((action-0 or action-1)) and action-2)"
    assert japanese == "(((action-0 または action-1)) かつ action-2)"

    responses = {f"action-{index}": _response(index) for index in range(3)}

    def parse_leaf(text, position):
        end = position
        while end < len(text) and not text[end].isspace() and text[end] != ")":
            end += 1
        return responses[text[position:end]], end

    parsed_english = parse_rendered_response_expression(
        english, parse_leaf, language="en")
    parsed_japanese = parse_rendered_response_expression(
        japanese, parse_leaf, language="ja")
    expected = canonical_response_expression_meaning(projected)
    assert canonical_response_expression_meaning(parsed_english) == expected
    assert canonical_response_expression_meaning(parsed_japanese) == expected
    assert response_expression_findings(parsed_english) == ()
    assert response_expression_findings(parsed_japanese) == ()
    with pytest.raises(ResponseExpressionError) as caught:
        render_response_expression(projected, render_leaf, language="unknown")
    assert caught.value.code == "RESPONSE_EXPRESSION_LANGUAGE_UNSUPPORTED"

    with pytest.raises(ResponseExpressionError) as caught:
        render_response_expression(projected, lambda response: "", language="ja")
    assert caught.value.code == "RESPONSE_EXPRESSION_RENDER_EMPTY"

    with pytest.raises(ResponseExpressionError) as caught:
        parse_rendered_response_expression("(action-0 かつ", parse_leaf, language="ja")
    assert caught.value.code == "RESPONSE_EXPRESSION_PARSE_INCOMPLETE"


def test_single_response_migration_and_rollback_are_lossless_and_schema_valid():
    _, base, candidate = _schemas()
    original = _document()
    assert list(Draft202012Validator(base).iter_errors(original)) == []
    migrated = migrate_requirement_to_response_expression(original)
    assert list(Draft202012Validator(candidate).iter_errors(migrated)) == []
    assert migrated["rimay_core"]["response_expression"]["kind"] == "response"
    assert rollback_requirement_from_response_expression(migrated) == original
    assert original["schema_version"] == "0.5.0"

    wrong_version = copy.deepcopy(original)
    wrong_version["schema_version"] = "0.4.0"
    with pytest.raises(ResponseExpressionError) as caught:
        migrate_requirement_to_response_expression(wrong_version)
    assert caught.value.code == "RESPONSE_EXPRESSION_MIGRATION_VERSION"


def test_composite_rollback_fails_closed_instead_of_dropping_responses():
    document = _document()
    document["schema_version"] = RESPONSE_EXPRESSION_CANDIDATE_VERSION
    document["rimay_core"] = {
        "response_expression": project_repository_response_expression(
            _repository_expression(), _response)}
    with pytest.raises(ResponseExpressionError) as caught:
        rollback_requirement_from_response_expression(document)
    assert caught.value.code == "RESPONSE_EXPRESSION_ROLLBACK_LOSSY"
    assert caught.value.path == "/rimay_core/response_expression"


def test_composite_can_be_quarantined_losslessly_for_reader_rollback():
    _, base, candidate = _schemas()
    document = _document()
    document["schema_version"] = RESPONSE_EXPRESSION_CANDIDATE_VERSION
    document["rimay_core"] = {
        "response_expression": project_repository_response_expression(
            _repository_expression(), _response)}
    assert list(Draft202012Validator(candidate).iter_errors(document)) == []
    quarantined = quarantine_composite_response_expression_for_rollback(document)
    assert list(Draft202012Validator(base).iter_errors(quarantined)) == []
    assert "response_expression" not in quarantined["rimay_core"]
    record = quarantined["extensions"]["additional_semantics"][0]
    assert record["kind"] == "RIMAY_JP_RESPONSE_EXPRESSION_QUARANTINE"
    assert record["response_expression_sha256"].startswith("sha256:")
    assert restore_quarantined_response_expression(quarantined) == document

    tampered = copy.deepcopy(quarantined)
    tampered["extensions"]["additional_semantics"][0]["response_expression"][
        "kind"] = "or"
    with pytest.raises(ResponseExpressionError) as caught:
        restore_quarantined_response_expression(tampered)
    assert caught.value.code == "RESPONSE_EXPRESSION_QUARANTINE_INVALID"


def test_store_replay_is_deterministic_and_does_not_mutate_inputs():
    documents = [_document("REQ-CANDIDATE-1"), _document("REQ-CANDIDATE-2")]
    original = copy.deepcopy(documents)
    first = replay_response_expression_migration(documents)
    second = replay_response_expression_migration(documents)
    assert first == second
    assert first["document_count"] == 2
    assert first["replay_sha256"].startswith("sha256:")
    assert documents == original
