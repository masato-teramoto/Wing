def _leaf(value: str = "Open") -> dict:
    return {"kind": "value", "value": value}


def _tree() -> dict:
    return {
        "kind": "or",
        "left": _leaf("Pending"),
        "right": {"kind": "group", "expression": {
            "kind": "and", "left": _leaf("1000"), "right": _leaf("Approved")}},
    }

               value_expression: dict | None = None) -> dict:
    return {"type": "IF", "predicate": {
        "operand": copy.deepcopy(operand), "operator": operator,
        "value_expression": copy.deepcopy(value_expression or _leaf()),
    }}

def test_candidate_schema_and_runtime_shape_agree():
    schema = json.loads(SCHEMA.read_text(encoding="utf-8"))
    envelope = {
        "candidate_version": OPERATOR_VALUE_EXPRESSION_CANDIDATE_VERSION,
        "target_canonical_version": OPERATOR_VALUE_EXPRESSION_TARGET_CANONICAL_VERSION,

def _assignment(owner: str, path: list[str], feature: str, start: int, end: int,
                source: str, values: list[dict]) -> dict:
    return {
        "owner_rule": owner, "rule_path": path, "feature": feature,
        "operator": "=", "span": {"start": start, "end": end},
        "source": source, "values": values,
    }


def _terminal(value: str, start: int, end: int, terminal: str = "STRING") -> dict:
    return {"kind": "terminal", "terminal": terminal, "value": value,
            "surface": f'"{value}"' if terminal == "STRING" else value,
            "span": {"start": start, "end": end}}


def _manual_repository(wrapper: str) -> tuple[dict, str, SimpleNamespace]:
    if wrapper == "ActorOrTypeOrPropertyOperatorElement":
        source = 'Order is "Open"'
        operand_end = 5
        assignments = [_assignment(
            wrapper, [wrapper], "actor_or_type_or_property", 0, operand_end, "Order",
            [{"kind": "cross_reference", "target": "ActorOrClassOrProperty",
              "terminal": "IDPointID", "value": "Order", "surface": "Order",
              "span": {"start": 0, "end": operand_end}, "linking_checked": True}])]
        symbols = ({"declaration_rule": "Class", "name": "Order",
                    "qualified_name": "Order", "span": {"start": 20, "end": 25},
                    "tree_path": "/declaration", "symbol_id": "SYM-001"},)
        references = ({"target": "ActorOrClassOrProperty", "terminal": "IDPointID",
                       "surface": "Order", "span": {"start": 0, "end": operand_end},
                       "status": "resolved", "compatible_declaration_rules":
                           ["Actor", "Class", "PropertyId"],
                       "candidate_symbol_ids": ["SYM-001"],
                       "resolved_symbol_id": "SYM-001"},)
        value_start = 9
    elif wrapper == "InstanceOrPropertyOperatorValue":
        source = '"order" is "Open"'
        operand_end = 7
        assignments = [_assignment(
            wrapper, [wrapper], "non_ui_instance_or_property", 0, operand_end,
            '"order"', [_terminal("order", 0, operand_end)])]
        symbols = ()
        references = ()
        value_start = 11
    else:
        source = '"submit" Button is "Open"'
        operand_end = 8
        assignments = [
            _assignment(wrapper, [wrapper], "ui_component_name", 0, operand_end,
                        '"submit"', [_terminal("submit", 0, operand_end)]),
            _assignment(wrapper, [wrapper], "ui_component_type", 9, 15, "Button",
                        [{"kind": "cross_reference", "target": "UIComponent",
                          "terminal": "ID", "value": "Button", "surface": "Button",
                          "span": {"start": 9, "end": 15}, "linking_checked": True}]),
        ]
        symbols = ({"declaration_rule": "UIComponent", "name": "Button",
                    "qualified_name": "Button", "span": {"start": 30, "end": 36},
                    "tree_path": "/declaration", "symbol_id": "SYM-001"},)
        references = ({"target": "UIComponent", "terminal": "ID",
                       "surface": "Button", "span": {"start": 9, "end": 15},
                       "status": "resolved", "compatible_declaration_rules": ["UIComponent"],
                       "candidate_symbol_ids": ["SYM-001"],
                       "resolved_symbol_id": "SYM-001"},)
        value_start = 19
    operator_start = value_start - 3
    value_end = len(source)
    assignments.extend([
        _assignment("OperatorValuesExpr", [wrapper, "OperatorValuesExpr"],
                    "operator", operator_start, operator_start + 2, "is",
                    [{"kind": "literal", "value": "is", "surface": "is",
                      "span": {"start": operator_start, "end": operator_start + 2}}]),
        _assignment("Compare", [wrapper, "OperatorValuesExpr", "Compare"],
                    "is_are", operator_start, operator_start + 2, "is",
                    [{"kind": "literal", "value": "is", "surface": "is",
                      "span": {"start": operator_start, "end": operator_start + 2}}]),
        _assignment("OperatorValuesExpr", [wrapper, "OperatorValuesExpr"],
                    "valuesExpression", value_start, value_end, '"Open"',
                    [_terminal("Open", value_start, value_end)]),
        _assignment("AtomicValue",
                    [wrapper, "OperatorValuesExpr", "MultiValuesExpr", "ValueOrValue",
                     "ValueAndValue", "ValuePrimaryExpression", "AtomicValue"],
                    "value", value_start, value_end, '"Open"',
                    [_terminal("Open", value_start, value_end)]),
    ])
    rule = {"rule_id": wrapper, "span": {"start": 0, "end": len(source)},
            "source": source, "assignments": assignments}
    linking = SimpleNamespace(
        succeeded=True, grammar_source_sha256=GRAMMAR_SHA256,
        symbols=symbols, references=references)
    return rule, source, linking

def test_active_migration_requires_explicit_matching_type_and_rollback_quarantines():
    active = {"type": "IF", "predicate": {
        "subject": "Order", "operator": "IS", "value": "Open"}}
    operand = {"wrapper": "declared_reference", "kind": "class", "name": "Order"}
    candidate = migrate_active_condition(active, operand)
    assert candidate == _condition(operand)
    with pytest.raises(OperatorValueExpressionCandidateError) as caught:
        rollback_active_condition(candidate)
    assert caught.value.code == \
        "OPERATOR_VALUE_CANDIDATE_ROLLBACK_REQUIRES_QUARANTINE"
    quarantine = quarantine_operator_value_condition(candidate)
    assert restore_operator_value_quarantine(quarantine) == candidate
    tampered = copy.deepcopy(quarantine)
    tampered["payload"]["predicate"]["operand"]["kind"] = "property"
    with pytest.raises(OperatorValueExpressionCandidateError) as tamper_error:
        restore_operator_value_quarantine(tampered)
    assert tamper_error.value.code == "OPERATOR_VALUE_CANDIDATE_QUARANTINE_TAMPERED"

def _fixed_projection(grammar, text: str):
    parsed = grammar.recognizer().parse_syntax(
        text, start_rule="CNLFunctionalRequirements")
    assert parsed.accepted and parsed.ambiguity_detected is False
    linking = link_repository_syntax(grammar, parsed, text)
    assert linking.succeeded, linking.problems
    projection = project_repository_document(parsed, text, linking, grammar)
    assert projection.status == "projected_document", projection.reason
    assert repository_document_problems(projection.document, text) == []
    condition = projection.document["requirements"][0]["condition_structure"] \
        ["conditions"][0]["condition"]
    rule = condition["payload"]["expression"]["rule"]
    return parsed, linking, rule


@pytest.mark.grammar
def test_fixed_repository_all_three_wrappers_project_with_linked_identity():
    path = os.environ.get("RIMAY_FIXED_GRAMMAR")
    if not path:
        pytest.skip("RIMAY_FIXED_GRAMMAR 未指定")
    grammar = compile_fixed_grammar(path)
    declarations = (
        "Actors: actor SystemA "
        "Classes: class Order := status "
        "UI_Component_Types: Button "
    )
    cases = (
        ('If Order.status is "Open" OR "Pending"', "property"),
        ('If "order" of type Order (referred to as CurrentOrder) is "Open"',
         "instance_or_property"),
        ('If "confirm" Button is "enabled"', "ui_component_instance"),
    )
    for surface, expected_kind in cases:
        text = declarations + surface + ", SystemA must calculate."
        parsed, linking, rule = _fixed_projection(grammar, text)
        candidate = project_repository_operator_value_condition(
            rule, text, linking, condition_type="IF",
            grammar_source_sha256=parsed.grammar_source_sha256)
        assert candidate["predicate"]["operand"]["kind"] == expected_kind
        assert operator_value_semantics(parse_operator_value_condition(
            render_operator_value_condition(candidate, language="ja"), language="ja"
        )) == operator_value_semantics(candidate)


@pytest.mark.grammar
def test_fixed_repository_modifier_and_free_expression_residuals_fail_closed():
    path = os.environ.get("RIMAY_FIXED_GRAMMAR")
    if not path:
        pytest.skip("RIMAY_FIXED_GRAMMAR 未指定")
    grammar = compile_fixed_grammar(path)
    cases = (
        ('If the "order" is "Open"', "OPERATOR_VALUE_CANDIDATE_MODIFIER_DEFERRED"),
        ('If "order" is a "Open"', "OPERATOR_VALUE_CANDIDATE_MODIFIER_DEFERRED"),
        ('If "order" is "Open" "recently"',
         "OPERATOR_VALUE_CANDIDATE_FREE_EXPRESSION_DEFERRED"),
    )
    for surface, code in cases:
        text = "Actors: actor SystemA " + surface + ", SystemA must calculate."
        parsed, linking, rule = _fixed_projection(grammar, text)
        with pytest.raises(OperatorValueExpressionCandidateError) as caught:
            project_repository_operator_value_condition(
                rule, text, linking, condition_type="IF",
                grammar_source_sha256=parsed.grammar_source_sha256)
        assert caught.value.code == code
