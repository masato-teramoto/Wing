#!/usr/bin/env python3
"""Atomic ResponseのCanonical 0.9.0隔離candidateを決定的に生成・検査する。"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "core"))

from rimay_jp_core.atomic_response_candidate import (  # noqa: E402
    build_atomic_response_candidate_schema,
)


BASE = ROOT / "canonical/canonical-requirement.schema.json"
OUTPUT = (
    ROOT / "canonical/candidates/"
    "canonical-requirement-atomic-response-0.9.0.schema.json"
)

_LEGACY_DESCRIPTION = (
    "Rimay-JP Canonical Requirement v0.8.0。unknownは値として格納せず、"
    "値なし(null/欠落)＋resolution.slotsの状態で表す。状態・根拠は意味を持つ"
    "リーフスロットへ付与し、複合オブジェクトの状態は子から導出する。"
    "response、response_expression、response_blockと、conditions、condition_expressionは"
    "それぞれ相互排他の互換形・拡張形であり、二重正本を禁止する。response_blockは"
    "item順、順次実行、bullet、connector、Atomic Responseを保持する。"
    "repository_traceは意味比較から除外するRepository provenanceである。"
)


def historical_base_schema(active: dict) -> dict:
    """Active 0.9.0からcandidate生成時の0.8.0 baseを再構築する。"""
    import copy
    base = copy.deepcopy(active)
    base["description"] = _LEGACY_DESCRIPTION
    base["properties"]["schema_version"]["const"] = "0.8.0"
    base["$defs"].pop("responseFrequency", None)
    base["$defs"].pop("repositoryAtomicResponseTrace", None)
    response = base["$defs"]["response"]
    response["properties"].pop("frequency", None)
    response["properties"].pop("repository_trace", None)
    return base


def rendered_candidate() -> str:
    with BASE.open(encoding="utf-8") as stream:
        active = json.load(stream)
    candidate = build_atomic_response_candidate_schema(historical_base_schema(active))
    return json.dumps(candidate, ensure_ascii=False, indent=2) + "\n"


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--check", action="store_true")
    args = parser.parse_args()
    rendered = rendered_candidate()
    if args.check:
        if not OUTPUT.is_file() or OUTPUT.read_text(encoding="utf-8") != rendered:
            raise SystemExit("atomic response candidate Schema is stale")
        print("atomic response candidate Schema: PASS")
        return
    OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT.write_text(rendered, encoding="utf-8")
    print(OUTPUT.relative_to(ROOT))


if __name__ == "__main__":
    main()
