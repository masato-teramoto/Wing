GRAMMAR = '''grammar example.Canonical with org.eclipse.xtext.common.Terminals
generate canonical "https://example.invalid/canonical"
CNLFunctionalRequirements: ('Actors:' actors+=Actor*)? requirements+=Requirement*;
Actor: 'actor' name=ID;
Requirement: (scope=Scope)? pattern=(WithoutCondition|WithCondition) '.';
Scope: 'For' scope_text+=ScopeText ('and' scope_text+=ScopeText)? ',';
ScopeText: scope+=STRING;
WithoutCondition: system_response=SystemResponse;
WithCondition: condition+=ConditionStructure
  (connector+=('and'|'or') condition+=ConditionStructure)*
  ',' system_response=SystemResponse;
ConditionStructure: IfStructure | WhileStructure | WhenStructure | TemporalStructure;
IfStructure: keyword='If' conditions=PreconditionStructure;
WhileStructure: keyword='While' preconditions=PreconditionStructure;
WhenStructure: keyword='When' trigger=(Trigger|FreeExpression);
TemporalStructure: RepeatOnce;
RepeatOnce: advice_keyword=('Before'|'After') event=Trigger;
PreconditionStructure: condition=Condition;
Condition: value=InstanceOrPropertyOperatorValue;
InstanceOrPropertyOperatorValue:
  (modifier=Modifier)? non_ui_instance_or_property=STRING rest=OperatorValuesExpr;
OperatorValuesExpr: operator=(Compare|Contains|OtherOperators)
  (value_modifier+=Modifier)? valuesExpression=MultiValuesExpr
  (additional_information=FreeExpression)?;
Compare: is_are='is' (not='not')?
  comparison=(EqualsTo|GreaterThan|LessThan|GreaterOrEqualTo|LessOrEqualTo)?;
EqualsTo: 'equal' 'to';
GreaterThan: 'greater' 'than';
LessThan: 'less' 'than';
GreaterOrEqualTo: 'greater' 'than' 'or' 'equal' 'to';
LessOrEqualTo: 'less' 'than' 'or' 'equal' 'to';
Contains: contains=('contains'|'has') negation=Negation?;
Negation: 'not';
OtherOperators: 'is' 'available';
MultiValuesExpr: values=ValueOrValue;
ValueOrValue returns MultiValuesExpr:
  ValueAndValue ({ValueOrValue.multileft=current} ('OR') right=ValueAndValue)*;
ValueAndValue returns MultiValuesExpr:
  ValuePrimaryExpression ({ValueAndValue.multileft=current} ('AND') right=ValuePrimaryExpression)*;
ValuePrimaryExpression returns MultiValuesExpr:
  '(' MultiValuesExpr ')' | value=AtomicValue;
AtomicValue: value=STRING;
FreeExpression: free_format_text=STRING;
Modifier: article=('a'|'an'|'the');
Trigger: actor=[Actor|ID] actions=ActionsExpression;
ActionsExpression: expression=AtomicActionInTrigger;
AtomicActionInTrigger: action=ActionPhrase;
SystemResponse:
  system_name=[Actor|ID] modal_verb=ModalVerb response_block=AtomicSystemResponse;
enum ModalVerb: SHALL='shall' | MUST='must';
AtomicSystemResponse: ('not')? system_response=ActionPhrase;
ActionPhrase: CREATE_26_4 | UnsupportedAction;
CREATE_26_4: verb='compute' property=STRING ('using' artifact=STRING)?;
UnsupportedAction: 'frobnicate' value=STRING;
'''


def _project(text: str):
    grammar = compile_xtext_grammar(GRAMMAR)
    parsed = grammar.recognizer().parse_syntax(
        text, start_rule="CNLFunctionalRequirements")
    assert parsed.accepted
    linked = link_repository_syntax(grammar, parsed, text)
    assert linked.succeeded
    projection = default_repository_adapter_registry().project(
        parsed, text, linked, grammar)
    return grammar, parsed, linked, projection


def test_builtin_adapter_projects_a_linked_calculate_requirement_to_valid_canonical():
    text = 'Actors: actor SystemA SystemA must compute "Result" using "Input".'
    _, parsed, linked, projection = _project(text)
    assert projection.status == "adapted_candidate"
    assert projection.unresolved_paths == ()
    canonical = projection.canonical_candidate
    assert canonical["source"]["text"] == text
    assert canonical["rimay_core"] == {
        "scope": None,
        "conditions": [],
        "response": {
            "actor": "SystemA", "modality": "MUST",
            "action": {"family": "CALCULATE", "rimay_action_id": "CREATE_26_4"},
            "result": "Result", "input": "Input",
        },
    }
    assert validate_schema(canonical) == []
    assert not any(item["blocking"] for item in validate(canonical, phase2_slice=False))
    assert canonical["requirement_id"].startswith("REQ-REPOSITORY-")
    assert parsed.grammar_source_sha256 == linked.grammar_source_sha256
    assert not parsed.grammar_source_sha256.startswith("sha256:sha256:")

def test_repository_projection_has_distinct_provenance_from_profile_conformance():
    text = ('Actors: actor SystemA '
            'If "Amount" is greater than "100", '
            'SystemA must compute "Result" using "Input".')
    _, _, _, projection = _project(text)
    canonical = projection.canonical_candidate
    mapped_paths = {
        "/rimay_core/conditions/0/type",
        "/rimay_core/response/action/family",
        "/rimay_core/response/action/rimay_action_id",
    }
    provenance = {item["path"]: item for item in canonical["provenance"]}
    assert {
        (provenance[path]["resolution_method"], provenance[path].get("reference"))
        for path in mapped_paths
    } == {("rule_application", REPOSITORY_GRAMMAR_PROJECTION_REFERENCE)}
    assert {
        item["path"] for item in canonical["provenance"]
        if item.get("reference") == REPOSITORY_GRAMMAR_PROJECTION_REFERENCE
    } == mapped_paths
    assert all(item["resolution_method"] != "profile_mapping"
               for item in canonical["provenance"])
    assert all(item["resolution_method"] != "repository_grammar_projection"
               for item in canonical["provenance"])
    assert validate_schema(canonical) == []

@pytest.mark.grammar
def test_builtin_adapter_is_source_bound_to_fixed_repository_grammar():
    path = os.environ.get("RIMAY_FIXED_GRAMMAR")
    if not path:
        pytest.skip("RIMAY_FIXED_GRAMMAR 未指定")
    grammar = compile_fixed_grammar(path)
    assert set(ACTION_ROLE_PATHS) <= set(grammar.by_name)

    def walk_ir(node):
        yield node
        for child in node.get("children", []):
            yield from walk_ir(child)
        if isinstance(node.get("child"), dict):
            yield from walk_ir(node["child"])

    for rule, roles in ACTION_ROLE_PATHS.items():
        pending, seen, actual_features = [rule], set(), set()
        while pending:
            current = pending.pop()
            if current in seen or current not in grammar.by_name:
                continue
            seen.add(current)
            for node in walk_ir(grammar.by_name[current].expression):
                if node["kind"] == "assignment":
                    actual_features.add(node["feature"])
                elif node["kind"] == "rule_ref":
                    pending.append(node["name"])
        assert {feature for paths in roles.values() for path in paths for feature in path} <= \
            actual_features, rule
    cases = [
        ('SystemA must pass "Invoice" to "Customer".',
         "SEND", "SEND_11_1_1", {"theme": "Invoice", "destination": "Customer"}),
        ('SystemA must update "Prop" into "Val".',
         "UPDATE", "DS_UPDATE", {"property": "Prop", "result_value": "Val"}),
        ('SystemA must generate "Invoice" in "OrderData".',
         "CREATE", "ENGENDER_27", {"theme": "Invoice"}),
        ('SystemA must store value "AuditLog" in "DB".',
         "STORE", "KEEP_15_2", {"theme": "AuditLog", "location": "DB"}),
        ('SystemA must accept "OrderData" from "Sender".',
         "RECEIVE", "OBTAIN_13_5_2", {"theme": "OrderData", "source": "Sender"}),
        ('SystemA must display "OrderList" on "Screen" Widget.',
         "DISPLAY", "REFLEXIVE_APPEARANCE_48_1_2",
         {"theme": "OrderList", "display_location": "Screen"}),
        ('SystemA must delete "AuditLog".',
         "DELETE", "REMOVE_10_1", {"theme": "AuditLog"}),
        ('SystemA must validate "Signature" by checking that following "Rule".',
         "VERIFY", "DS_VALIDATE", {"theme": "Signature", "rule": "Rule"}),
        ('SystemA must compute "Result" using "Input".',
         "CALCULATE", "CREATE_26_4", {"result": "Result", "input": "Input"}),
        ('SystemA must notify "Notice" to "Customer".',
         "NOTIFY", "SAY_37_7", {"content": "Notice", "destination": "Customer"}),
    ]
    for requirement, family, rule, roles in cases:
        declarations = "Actors: actor SystemA\n"
        if family == "DISPLAY":
            declarations = "UI_Component_Types: Widget\n" + declarations
        text = declarations + requirement
        parsed = grammar.recognizer().parse_syntax(
            text, start_rule="CNLFunctionalRequirements")
        linked = link_repository_syntax(grammar, parsed, text)
        projection = default_repository_adapter_registry().project(
            parsed, text, linked, grammar)
        assert parsed.accepted and linked.succeeded, requirement
        assert projection.status == "adapted_candidate", requirement
        assert projection.unresolved_paths == (), requirement
        canonical = projection.canonical_candidate
        assert canonical["rimay_core"]["response"] == {
            "actor": "SystemA", "modality": "MUST",
            "action": {"family": family, "rimay_action_id": rule}, **roles,
        }, requirement
        assert validate_schema(canonical) == [], requirement
        assert not any(item["blocking"] for item in
                       validate(canonical, phase2_slice=False)), requirement

@pytest.mark.grammar
def test_all_fixed_repository_action_dispatches_preserve_the_selected_rule_id():
    """The 50-rule dispatch population is exact, not a representative sample."""
    path = os.environ.get("RIMAY_FIXED_GRAMMAR")
    if not path:
        pytest.skip("RIMAY_FIXED_GRAMMAR 未指定")
    grammar = compile_fixed_grammar(path)

    def walk_ir(node):
        yield node
        for child in node.get("children", []):
            yield from walk_ir(child)
        if isinstance(node.get("child"), dict):
            yield from walk_ir(node["child"])

    direct_actions = sorted(
        node["name"] for node in walk_ir(grammar.by_name["ActionPhrase"].expression)
        if node["kind"] == "rule_ref")
    assert len(direct_actions) == len(set(direct_actions)) == 50

    recognizer = grammar.recognizer()
    for rule in direct_actions:
        surface = recognizer.synthesize_minimal(rule)
        parsed = recognizer.parse_syntax(surface, start_rule="ActionPhrase")
        assert parsed.accepted and parsed.ambiguity_detected is False, rule
        assert [node["name"] for node in direct_syntax_rule_children(parsed.tree)] == [rule]
