#!/usr/bin/env python3
"""Itemized Response Blockのaccepted Canonical 0.8.0候補を決定的に検査する。

Active Schema昇格後もcandidate生成証跡を再現できるよう、Response Block差分を
除いた歴史的0.7.0 baseを決定的に再構築する。
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "core"))

from rimay_jp_core.response_block_candidate import (  # noqa: E402
    build_response_block_candidate_schema,
)


BASE = ROOT / "canonical/canonical-requirement.schema.json"
OUTPUT = (
    ROOT / "canonical/candidates/"
    "canonical-requirement-response-block-0.8.0.schema.json"
)

_LEGACY_DESCRIPTION = (
    "Rimay-JP Canonical Requirement v0.7.0。unknownは値として格納せず、"
    "値なし(null/欠落)＋resolution.slotsの状態で表す。状態・根拠は意味を持つ"
    "リーフスロットへ付与し、複合オブジェクトの状態は子から導出する。"
    "responseとresponse_expression、conditionsとcondition_expressionはそれぞれ"
    "互換形と式木拡張形であり、二重正本を禁止する。condition_expressionは"
    "condition leafとAND/OR木を保持し、repository_traceは意味比較から除外する"
    "原典provenanceである。"
)
_BLOCK_DEFS = (
    "repositoryResponseBlockTrace", "repositoryResponseBlockItemTrace",
    "responseBlockBullet", "responseBlockItem", "responseBlock",
)
_POST_BLOCK_DEFS = ("responseFrequency", "repositoryAtomicResponseTrace")


def historical_base_schema(active: dict) -> dict:
    """Active 0.8.0からaccepted candidate生成時の0.7.0 baseを再構築する。"""
    import copy
    base = copy.deepcopy(active)
    base["description"] = _LEGACY_DESCRIPTION
    base["properties"]["schema_version"]["const"] = "0.7.0"
    core = base["$defs"]["rimayCore"]
    core["properties"].pop("response_block", None)
    removed = {
        ("response", "response_block"),
        ("response_expression", "response_block"),
    }
    core["allOf"] = [
        item for item in core.get("allOf", [])
        if tuple((item.get("not") or {}).get("required") or []) not in removed
    ]
    response = base["$defs"]["response"]
    response["properties"].pop("frequency", None)
    response["properties"].pop("repository_trace", None)
    for name in _BLOCK_DEFS + _POST_BLOCK_DEFS:
        base["$defs"].pop(name, None)
    return base


def rendered_candidate() -> str:
    with BASE.open(encoding="utf-8") as stream:
        active = json.load(stream)
    base = historical_base_schema(active)
    candidate = build_response_block_candidate_schema(base)
    return json.dumps(candidate, ensure_ascii=False, indent=2) + "\n"


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--check", action="store_true")
    args = parser.parse_args()
    rendered = rendered_candidate()
    if args.check:
        if not OUTPUT.is_file() or OUTPUT.read_text(encoding="utf-8") != rendered:
            raise SystemExit("response block candidate Schema is stale")
        print("response block candidate Schema: PASS")
        return
    OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT.write_text(rendered, encoding="utf-8")
    print(OUTPUT.relative_to(ROOT))


if __name__ == "__main__":
    main()
