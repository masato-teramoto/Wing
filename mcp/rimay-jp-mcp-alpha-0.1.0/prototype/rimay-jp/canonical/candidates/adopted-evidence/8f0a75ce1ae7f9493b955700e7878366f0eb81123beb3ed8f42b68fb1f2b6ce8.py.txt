    RIMAY_JP / "canonical/candidates/"
    "canonical-requirement-atomic-response-0.9.0.schema.json"
)
SOURCE = 'SystemA must not compute "Result" every "3 seconds".'


def _span(fragment, *, start=0):
    begin = SOURCE.index(fragment, start)
    return {"start": begin, "end": begin + len(fragment)}


def _fact(value, surface, kind="literal"):
    fact = {"kind": kind, "value": value, "surface": surface, "span": _span(surface)}
    if kind == "enum_literal":
        fact["name"] = value.upper()
    elif kind == "terminal":
        fact["terminal"] = "STRING"
    return fact


def _option(value, surface, kind="literal"):
    return {"span": _span(surface), "source": surface,
            "value": _fact(value, surface, kind)}


def _system_modal(value="must"):
    surface = value
    return {
        "owner_rule": "SystemResponse", "rule_path": ["SystemResponse"],
        "feature": "modal", "operator": "=", "span": _span(surface),
        "source": surface, "values": [_fact(value, surface, "enum_literal")],
    }


def _frequency(expression="3 seconds"):
    full = f'every "{expression}"'
    return {
        "span": _span(full), "source": full,
        "keyword": {"value": "every", "span": _span("every"), "source": "every"},
        "value": _fact(expression, f'"{expression}"', "terminal"),
    }


def _repository_atomic(*, local_modal=None, negated=True, frequency=True):
    start = SOURCE.index("not") if negated else SOURCE.index("compute")
    return {
        "index": 0,
        "span": {"start": start, "end": SOURCE.index(".")},
        "source": SOURCE[start:SOURCE.index(".")],
        "modality": None if local_modal is None else _option(
            local_modal, local_modal, "enum_literal"),
        "negation": _option("not", "not") if negated else None,
        "frequency": _frequency() if frequency else None,
        "action": {"rule_id": "CREATE_26_4", "span": _span('compute "Result"'),
                   "source": 'compute "Result"', "assignments": []},
    }


def _action(_record):
    return {
        "actor": "SystemA",
        "action": {"family": "CALCULATE", "rimay_action_id": "CREATE_26_4"},
        "result": "Result", "input": "Input",
    }


def _projected():
    return project_repository_atomic_response(
        _repository_atomic(), _system_modal(), SOURCE, _action)


def _document(requirement_id="REQ-ATOMIC-CANDIDATE"):
    document = new_canonical(requirement_id)
    document["source"]["text"] = SOURCE
    document["rimay_core"]["response"] = {
        **_action({}), "modality": "MUST",
    }
    return document


def _schemas():
    active = json.loads(ACTIVE_SCHEMA.read_text(encoding="utf-8"))
    candidate = json.loads(CANDIDATE_SCHEMA.read_text(encoding="utf-8"))
    return active, candidate


def test_candidate_schema_remains_deterministic_after_active_0_9_activation():
    active, candidate = _schemas()
    historical = historical_base_schema(active)
    assert candidate == build_atomic_response_candidate_schema(historical)
    assert CANDIDATE_SCHEMA.read_text(encoding="utf-8") == rendered_candidate()
    Draft202012Validator.check_schema(candidate)
    assert historical["properties"]["schema_version"]["const"] == \
        ACTIVE_CANONICAL_VERSION
    assert active["properties"]["schema_version"]["const"] == \
        ATOMIC_RESPONSE_CANDIDATE_VERSION
    assert candidate["properties"]["schema_version"]["const"] == \
        ATOMIC_RESPONSE_CANDIDATE_VERSION
    assert "frequency" not in historical["$defs"]["response"]["properties"]
    assert "frequency" in active["$defs"]["response"]["properties"]
    assert "frequency" in candidate["$defs"]["response"]["properties"]
    assert "repository_trace" in candidate["$defs"]["response"]["properties"]


def test_projection_inherits_wrapper_modal_and_keeps_negation_frequency_and_trace():
    response = _projected()
    assert response["modality"] == "MUST_NOT"
    assert response["frequency"] == {
        "value": 3, "unit": "second", "source_expression": "3 seconds"}
    trace = response["repository_trace"]
    assert trace["repository_rule_id"] == "AtomicSystemResponse"
    assert trace["atomic_response_index"] == 0
    assert trace["modality_origin"] == "system_response"
    assert trace["modality_source"] == "must"
    assert trace["negation_source"] == "not"
    assert trace["frequency_source"] == 'every "3 seconds"'
    assert atomic_response_findings(response, source_text=SOURCE) == ()


def test_local_shall_override_is_normalized_without_losing_its_surface_origin():
    source = SOURCE.replace("not compute", "shall compute")
    atomic = _repository_atomic(negated=False, frequency=False)
    start = source.index("shall")
    atomic.update({
        "span": {"start": start, "end": source.index(".")},
        "source": source[start:source.index(".")],
        "modality": {
            "span": {"start": start, "end": start + 5}, "source": "shall",
            "value": {"kind": "enum_literal", "name": "SHALL", "value": "shall",
                      "surface": "shall", "span": {"start": start, "end": start + 5}},
        },
        "action": {"rule_id": "CREATE_26_4",
                   "span": {"start": source.index("compute"),
                            "end": source.index("compute") + len('compute "Result"')},
                   "source": 'compute "Result"', "assignments": []},
    })
    modal = copy.deepcopy(_system_modal())
    response = project_repository_atomic_response(atomic, modal, source, _action)
    assert response["modality"] == "MUST"
    assert response["repository_trace"]["modality_origin"] == "atomic_response"
    assert response["repository_trace"]["modality_source"] == "shall"


def test_candidate_schema_accepts_leaf_local_frequency_in_all_response_containers():
    active, candidate = _schemas()
    document = _document()
    document["schema_version"] = ATOMIC_RESPONSE_CANDIDATE_VERSION
    document["rimay_core"]["response"] = _projected()
    assert list(Draft202012Validator(candidate).iter_errors(document)) == []
    assert list(Draft202012Validator(active).iter_errors(document)) == []

    expression = copy.deepcopy(document)
    response = expression["rimay_core"].pop("response")
    expression["rimay_core"]["response_expression"] = {
        "kind": "response", "response": response,
        "repository_trace": {
            "repository_rule_id": "AtomicSystemResponse",
            "source_span": response["repository_trace"]["source_span"],
            "source": response["repository_trace"]["source"],
            "atomic_response_index": 0,
        },
    }
    assert list(Draft202012Validator(candidate).iter_errors(expression)) == []

    block = copy.deepcopy(document)
    response = block["rimay_core"].pop("response")
    block["rimay_core"]["response_block"] = {
        "kind": "itemized", "in_sequence": False,
        "items": [{"position": 0, "bullet": {"kind": "hyphen", "value": "-"},
                   "connector_before": None, "response": response}],
    }
    assert list(Draft202012Validator(candidate).iter_errors(block)) == []


def test_projection_and_diagnostics_fail_closed_on_ambiguity_and_tampering():
    changed = _repository_atomic()
    changed["frequency"]["value"]["value"] = "often"
    with pytest.raises(AtomicResponseError) as caught:
        project_repository_atomic_response(changed, _system_modal(), SOURCE, _action)
    assert caught.value.code == "ATOMIC_RESPONSE_FREQUENCY_UNMEASURABLE"

    changed = _repository_atomic()
    changed["frequency"]["keyword"]["source"] = "each"
    with pytest.raises(AtomicResponseError) as caught:
        project_repository_atomic_response(changed, _system_modal(), SOURCE, _action)
    assert caught.value.code == "ATOMIC_RESPONSE_SOURCE_TAMPERED"

    with pytest.raises(AtomicResponseError) as caught:
        project_repository_atomic_response(
            _repository_atomic(), _system_modal(), SOURCE,
            lambda _record: {**_action({}), "modality": "MUST"})
    assert caught.value.code == "ATOMIC_RESPONSE_ACTION_AUTHORITY_CONFLICT"

    changed = _projected()
    changed["repository_trace"]["frequency_source"] = "every invented"
    changed["repository_trace"]["negation_span"] = None
    codes = {item["code"] for item in atomic_response_findings(
        changed, source_text=SOURCE)}
    assert {"ATOMIC_RESPONSE_SOURCE_TAMPERED",
            "ATOMIC_RESPONSE_TRACE_PAIR_MISMATCH",
            "ATOMIC_RESPONSE_NEGATION_MODALITY_MISMATCH"} <= codes


@pytest.mark.grammar
def test_fixed_repository_atomic_response_feeds_candidate_without_surface_reparse():
    path = os.environ.get("RIMAY_FIXED_GRAMMAR")
    if not path:
        pytest.skip("RIMAY_FIXED_GRAMMAR 未指定")
    text = (
        "Actors: actor SystemA "
        'SystemA must subscribe to "text" every "3 seconds".'
    )
    grammar = compile_fixed_grammar(path)
    parsed = grammar.recognizer().parse_syntax(
        text, start_rule="CNLFunctionalRequirements")
    assert parsed.accepted and parsed.ambiguity_detected is False
    linking = link_repository_syntax(grammar, parsed, text)
    assert linking.succeeded
    result = project_repository_document(parsed, text, linking, grammar)
    assert result.status == "projected_document", result.reason
    assert repository_document_problems(result.document, text) == []
    requirement = result.document["requirements"][0]
    response = project_repository_atomic_response(
        requirement["atomic_responses"][0],
        requirement["condition_structure"]["system_response"]["modality"],
        text,
        lambda action: {
            "actor": "SystemA",
            "action": {"family": "SUBSCRIBE", "rimay_action_id": action["rule_id"]},
            "theme": "text",
        },
    )
    assert response["modality"] == "MUST"
    assert response["frequency"]["value"] == 3
    assert response["repository_trace"]["modality_origin"] == "system_response"
    assert atomic_response_findings(response, source_text=text) == ()


def test_english_and_japanese_controlled_surfaces_roundtrip_one_meaning():
    response = _projected()
    render_action = lambda item: item["action"]["family"]
    english = render_atomic_response(response, render_action, language="en")
    japanese = render_atomic_response(response, render_action, language="ja")
    assert english.startswith("atomic response [modality=MUST_NOT; frequency=3 second]")
    assert japanese.startswith("原子応答［モダリティ=禁止；頻度=3秒］")

    def parse_action(text):
        assert text == "CALCULATE"
        return _action({})

    parsed_en = parse_rendered_atomic_response(english, parse_action, language="en")
    parsed_ja = parse_rendered_atomic_response(japanese, parse_action, language="ja")
    expected = canonical_atomic_response_meaning(response)
    assert canonical_atomic_response_meaning(parsed_en) == expected
    assert canonical_atomic_response_meaning(parsed_ja) == expected
    with pytest.raises(AtomicResponseError) as caught:
        parse_rendered_atomic_response("SystemA must calculate", parse_action, language="en")
    assert caught.value.code == "ATOMIC_RESPONSE_PARSE_SHAPE"


def test_migration_rollback_quarantine_restore_and_replay_are_deterministic():
    active, candidate = _schemas()
    original = _document()
    original["schema_version"] = ACTIVE_CANONICAL_VERSION
    original["validation_state"] = {
        "report_id": "VR-OLD", "blocking_findings": 0, "warning_findings": 0,
        "validated_canonical_hash": "sha256:old",
        "validator_registry_version": "0.3.0",
    }
    migrated = migrate_requirement_to_atomic_response_candidate(original)
    assert migrated["schema_version"] == ATOMIC_RESPONSE_CANDIDATE_VERSION
    assert "validation_state" not in migrated
    assert list(Draft202012Validator(candidate).iter_errors(migrated)) == []
    rolled_back = rollback_requirement_from_atomic_response_candidate(migrated)
    expected = copy.deepcopy(original)
    expected.pop("validation_state")
    assert rolled_back == expected

    adopted = copy.deepcopy(migrated)
    adopted["rimay_core"]["response"] = _projected()
    with pytest.raises(AtomicResponseError) as caught:
        rollback_requirement_from_atomic_response_candidate(adopted)
    assert caught.value.code == "ATOMIC_RESPONSE_ROLLBACK_LOSSY"
    quarantined = quarantine_atomic_responses_for_rollback(adopted)
    historical = historical_base_schema(active)
    assert list(Draft202012Validator(historical).iter_errors(quarantined)) == []
    assert restore_quarantined_atomic_responses(quarantined) == adopted

    tampered = copy.deepcopy(quarantined)
    tampered["extensions"]["additional_semantics"][0]["records"][0][
        "additions"]["frequency"]["value"] = 9
    with pytest.raises(AtomicResponseError) as caught:
        restore_quarantined_atomic_responses(tampered)
    assert caught.value.code == "ATOMIC_RESPONSE_QUARANTINE_INVALID"

    documents = [_document("REQ-ATOMIC-1"), _document("REQ-ATOMIC-2")]
    for document in documents:
        document["schema_version"] = ACTIVE_CANONICAL_VERSION
    before = copy.deepcopy(documents)
    assert replay_atomic_response_migration(documents) == \
        replay_atomic_response_migration(documents)
    assert documents == before
