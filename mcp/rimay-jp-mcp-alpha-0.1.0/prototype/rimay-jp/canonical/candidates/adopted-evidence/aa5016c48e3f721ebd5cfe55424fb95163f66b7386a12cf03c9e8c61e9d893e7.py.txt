def _response(family: str = "SEND") -> dict:
    if family == "DELETE":
        return {
            "actor": "OrderManagementSystem", "modality": "MUST_NOT",
            "action": {"family": "DELETE", "rimay_action_id": "REMOVE_10_1"},
            "theme": "InvoicePDF",
        }
    return {
        "actor": "NotificationSystem", "modality": "MUST",
        "action": {"family": "SEND", "rimay_action_id": "SEND_11_1_1"},
        "theme": "InvoicePDF", "destination": "CustomerEmailAddress",
    }


def _condition(kind: str) -> dict:
    if kind == "WHEN":
        return {
            "type": "WHEN",
            "event": {
                "actor": "User", "action": {"family": "PRESS"},
                "object": "ConfirmOrderButton",
            },
        }
    return {
        "type": "IF",
        "predicate": {
            "subject": "OrderStatus", "operator": "IS", "value": "Confirmed",
        },
    }


def _condition_expression() -> dict:
    return {
        "kind": "and",
        "left": {"kind": "condition", "condition": _condition("WHEN")},
        "right": {"kind": "condition", "condition": _condition("IF")},
    }


def _canonical(*, response_expression: bool = False) -> dict:
    canonical = new_canonical("REQ-CONDITION-EXPR-ACTIVE")
    canonical["source"]["text"] = "複合条件"
    canonical["rimay_core"].pop("conditions")
    canonical["rimay_core"]["condition_expression"] = _condition_expression()
    if response_expression:
        canonical["rimay_core"].pop("response")
        canonical["rimay_core"]["response_expression"] = {
            "kind": "or",
            "left": {"kind": "response", "response": _response()},
            "right": {"kind": "response", "response": _response("DELETE")},
        }
    else:
        canonical["rimay_core"]["response"] = _response()
    return canonical


def test_active_schema_accepts_condition_expression_and_rejects_dual_source():
    canonical = _canonical()
    assert SCHEMA_VERSION == "1.1.0"
    assert CONDITION_EXPRESSION_VERSION == "0.7.0"
    assert validate_schema(canonical) == []
    dual = copy.deepcopy(canonical)
    dual["rimay_core"]["conditions"] = []
    assert validate_schema(dual)
    assert condition_core_findings(dual["rimay_core"])[0]["code"] == \
        "CONDITION_EXPRESSION_DUAL_SOURCE"


def test_english_and_japanese_roundtrip_condition_and_response_trees_together():
    canonical = _canonical(response_expression=True)
    views = render_requirement_views(canonical)
    english = views["english_rimay"]
    assert english.count(" and ") == 1 and english.count(" or ") == 1
    assert views["manifest"]["conformance"]["condition_expression"] is True
    assert set(views["manifest"]["conformance"]["extension_features"]) == {
        "COMPARISON_OPERATOR", "CONDITION_COMBINATION", "RESPONSE_COMBINATION"}
    ast = parse(english)
    assert ast["condition_expression"]["kind"] == "and"
    assert ast["response_expression"]["kind"] == "or"
    assert compare_semantic(canonical, ast) == []
    assert compare_conformance(canonical, ast) == []

    japanese = render_rimay_jp(canonical)
    assert " かつ " in japanese and " または " in japanese
    assert project({"rimay_core": parse_rimay_jp(japanese)}) == project(canonical)

def test_trace_tampering_blocks_validation_before_leaf_consumers():
    canonical = _canonical()
    canonical["rimay_core"]["condition_expression"]["repository_trace"] = {
        "repository_rule_id": "ConditionsOrExpression",
        "source_span": {"start": 0, "end": 3},
        "source": "A or B",
    }
    findings = validate(canonical, phase2_slice=False)
    assert any(
        item["code"] == "JP-COORD" and item["blocking"] and
        (item.get("evidence") or {}).get("condition_expression_code") ==
        "CONDITION_EXPRESSION_TRACE_RULE_MISMATCH"
        for item in findings
    )


def test_explicit_migration_rebases_pointer_metadata_and_invalidates_snapshot():
    legacy = new_canonical("REQ-CONDITION-MIGRATION")
    legacy["schema_version"] = "0.6.0"
    legacy["rimay_core"]["conditions"] = [_condition("IF")]
    legacy["rimay_core"]["response"] = _response()
    old = "/rimay_core/conditions/0/predicate/value"
    new = "/rimay_core/condition_expression/condition/predicate/value"
    legacy["resolution"]["slots"] = {old: {"status": "unknown"}}
    legacy["provenance"] = [{"id": "PRV-1", "path": old}]
    legacy["findings"] = [{"code": "TEST", "path": old}]
    legacy["approval"] = {
        "approved_by": "reviewer", "approved_at": "2026-08-17",
        "approved_slots": [old],
    }
    legacy["validation_state"] = {
        "report_id": "VR-1", "blocking_findings": 0, "warning_findings": 0,
        "validated_canonical_hash": "sha256:old",
        "validator_registry_version": "0.3.0",
    }
    migrated = migrate_requirement_to_condition_expression(legacy)
    assert new in migrated["resolution"]["slots"]
    assert migrated["provenance"][0]["path"] == new
    assert migrated["findings"][0]["path"] == new
    assert migrated["approval"]["approved_slots"] == [new]
    assert "validation_state" not in migrated
    rolled_back = rollback_requirement_from_condition_expression(migrated)
    assert old in rolled_back["resolution"]["slots"]
    assert rolled_back["provenance"][0]["path"] == old


def test_store_reader_accepts_05_06_07_and_emits_only_active_shape(tmp_path):
    documents = []
    for version in ("0.5.0", "0.6.0"):
        legacy = new_canonical(f"REQ-STORE-{version.replace('.', '')}")
        legacy["schema_version"] = version
        legacy["rimay_core"]["conditions"] = [_condition("IF")]
        legacy["rimay_core"]["response"] = _response()
        legacy["validation_state"] = {
            "report_id": "VR-OLD", "blocking_findings": 0,
            "warning_findings": 0,
            "validated_canonical_hash": "sha256:old",
            "validator_registry_version": "0.3.0",
        }
        documents.append(legacy)
    documents.append(_canonical())
    records = [
        {"record_id": f"EXP-{index:06d}", "kind": "canonical_revision",
         "payload": {"canonical": canonical}}
        for index, canonical in enumerate(documents, 1)
    ]
    replay = migrate_experience_store_records(records)
    assert replay["canonical_count"] == 3
    migrated = [item["payload"]["canonical"] for item in replay["records"]]
    assert {item["schema_version"] for item in migrated} == {"1.1.0"}
    assert migrated[1]["rimay_core"]["conditions"][0]["type"] == "IF"
    assert all("validation_state" not in item for item in migrated[:2])
    assert migrated[2]["rimay_core"]["condition_expression"]["kind"] == "and"

    path = tmp_path / "store.jsonl"
    path.write_text("\n".join(json.dumps(item) for item in records) + "\n",
                    encoding="utf-8")
    assert all(
        item["payload"]["canonical"]["schema_version"] == "1.1.0"
        for item in ExperienceStore(path).canonical_revisions()
    )


def test_cli_condition_migration_rollback_and_active_store_replay(tmp_path, capsys):
    from adapters.cli.rimay_jp_cli import main

    legacy = new_canonical("REQ-CLI-CONDITION")
    legacy["schema_version"] = "0.6.0"
    legacy["rimay_core"]["conditions"] = [_condition("IF")]
    legacy["rimay_core"]["response"] = _response()
    source = tmp_path / "legacy.json"
    source.write_text(json.dumps(legacy), encoding="utf-8")
    assert main(["rimay_jp_cli.py", "migrate-condition-expression", str(source)]) == 0
    migrated = json.loads(capsys.readouterr().out)["canonical"]
    assert migrated["schema_version"] == "0.7.0"

    target = tmp_path / "active.json"
    target.write_text(json.dumps(migrated), encoding="utf-8")
    assert main(["rimay_jp_cli.py", "rollback-condition-expression", str(target)]) == 0
    assert json.loads(capsys.readouterr().out)["canonical"] == legacy

    store = tmp_path / "store.jsonl"
    store.write_text(json.dumps({
        "record_id": "EXP-000001", "kind": "canonical_revision",
        "payload": {"canonical": legacy},
    }) + "\n", encoding="utf-8")
    assert main(["rimay_jp_cli.py", "replay-canonical-store", str(store)]) == 0
    assert json.loads(capsys.readouterr().out)["canonical_count"] == 1


def test_downstream_consumers_traverse_all_condition_leaves():
    canonical = _canonical()
    paths = [path for path, _condition_value in iter_core_conditions(
        canonical["rimay_core"])]
    assert paths == [
        "/rimay_core/condition_expression/left/condition",
        "/rimay_core/condition_expression/right/condition",
    ]
    assert validate(canonical, phase2_slice=False) == []
    assert _in_mvp(canonical)
    assert {feature for feature in _features(canonical)
            if feature.startswith("cond:")} == {"cond:WHEN", "cond:IF"}
    routed = route(copy.deepcopy(canonical))
    state_routes = [item["slot"] for item in routed["routes"]
                    if item["source"] == "state_model"]
    assert state_routes == [
        "/rimay_core/condition_expression/right/condition/predicate/value"]
    validated = validate_all(copy.deepcopy(canonical))
    assert validated["blocked"] is False
    assert validated["jp_roundtrip_ok"] is True
    approved = copy.deepcopy(canonical)
    approved["status"] = "approved"
    generated = generate_criteria(approved)
    assert generated["ok"] is False
    assert generated["findings"][0]["path"] == "/rimay_core/condition_expression"
    dialogue = DialogueState(copy.deepcopy(canonical))
    assert dialogue.open_questions() == []
    assert dialogue.answer({"question_id": "Q1"}, "回答")["ok"] is False
